### 2.2. Express Server Skeleton (`src/server.ts`)

```ts
import express, { Request, Response } from 'express';
import multer from 'multer';
import dotenv from 'dotenv';
import { processPdf } from './pdfProcessor';
import { answerQuestion } from './qaEngine';

dotenv.config();
const app = express();
app.use(express.json());

const upload = multer({ dest: 'uploads/' });

app.post('/upload', upload.single('file'), async (req: Request, res: Response) => {
  if (!req.file) return res.status(400).json({ error: 'No PDF uploaded' });
  try {
    await processPdf(req.file.path, req.file.filename);
    res.json({ message: 'PDF indexed successfully' });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: 'Failed to process PDF' });
  }
});

app.post('/ask', async (req: Request, res: Response) => {
  const { question, pdfId } = req.body;
  if (!question || !pdfId) return res.status(400).json({ error: 'Missing fields' });

  try {
    const answer = await answerQuestion(question, pdfId);
    res.json({ answer });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: 'Failed to generate answer' });
  }
});

const PORT = process.env.PORT || 3001;
app.listen(PORT, () => console.log(`🚀 Server listening on ${PORT}`));
```

### 2.3. PDF Processing (`src/pdfProcessor.ts`)

```ts
import fs from 'fs';
import pdf from 'pdf-parse';
import { embedChunks, upsertVectors } from './vectorStore';
import { splitIntoChunks } from './chunker';

export async function processPdf(filePath: string, pdfId: string) {
  const dataBuffer = fs.readFileSync(filePath);
  const pdfData = await pdf(dataBuffer);
  const text = pdfData.text;

  // 1️⃣ Chunk the raw text (e.g., 500‑token windows)
  const chunks = splitIntoChunks(text, 500);

  // 2️⃣ Create embeddings for each chunk
  const embeddings = await embedChunks(chunks);

  // 3️⃣ Upsert into vector DB, tagging each vector with pdfId + chunk index
  await upsertVectors(embeddings, pdfId, chunks);
}
```

#### Chunker (`src/chunker.ts`)

```ts
export function splitIntoChunks(text: string, maxTokens: number): string[] {
  // Very simple token estimator: 1 token ≈ 4 characters (good enough for demo)
  const approxChars = maxTokens * 4;
  const chunks: string[] = [];

  let start = 0;
  while (start < text.length) {
    const end = Math.min(start + approxChars, text.length);
    chunks.push(text.slice(start, end));
    start = end;
  }
  return chunks;
}
```

### 2.4. Embedding & Vector Store (`src/vectorStore.ts`)

```ts
import { Configuration, OpenAIApi } from 'openai';
import { PineconeClient } from '@pinecone-database/pinecone';
import dotenv from 'dotenv';
dotenv.config();

const openai = new OpenAIApi(new Configuration({ apiKey: process.env.OPENAI_API_KEY }));
const pinecone = new PineconeClient();

await pinecone.init({
  environment: process.env.PINECONE_ENVIRONMENT!,
  apiKey: process.env.PINECONE_API_KEY!,
});
const index = pinecone.Index(process.env.PINECONE_INDEX!);

/** Embed an array of strings (chunks) */
export async function embedChunks(chunks: string[]): Promise<number[][]> {
  const response = await openai.createEmbedding({
    model: 'text-embedding-ada-002',
    input: chunks,
  });
  // @ts-ignore – response shape from OpenAI SDK
  return response.data.data.map((d: any) => d.embedding);
}

/** Upsert vectors with metadata */
export async function upsertVectors(
  embeddings: number[][],
  pdfId: string,
  chunks: string[]
) {
  const vectors = embeddings.map((vec, i) => ({
    id: `${pdfId}-${i}`,
    values: vec,
    metadata: {
      pdfId,
      chunkIndex: i,
      text: chunks[i],
    },
  }));

  await index.upsert({ upsertRequest: { vectors } });
}

/** Retrieve top‑k similar chunks for a query */
export async function querySimilar(
  query: string,
  pdfId: string,
  topK = 5
): Promise<Array<{ text: string; score: number }>> {
  const [{ embedding }] = await openai.createEmbedding({
    model: 'text-embedding-ada-002',
    input: [query],
  }).then(r => r.data.data);

  const result = await index.query({
    queryRequest: {
      vector: embedding,
      topK,
      filter: { pdfId }, // restrict to the same PDF
      includeMetadata: true,
    },
  });

  return result.matches.map(m => ({
    text: (m.metadata as any).text,
    score: m.score,
  }));
}
```

### 2.5. QA Engine (`src/qaEngine.ts`)

```ts
import { querySimilar } from './vectorStore';
import { Configuration, OpenAIApi } from 'openai';
import dotenv from 'dotenv';
dotenv.config();

const openai = new OpenAIApi(new Configuration({ apiKey: process.env.OPENAI_API_KEY }));

export async function answerQuestion(question: string, pdfId: string) {
  // 1️⃣ Retrieve relevant chunks
  const relevant = await querySimilar(question, pdfId, 5);
  const context = relevant.map(r => r.text).join('\n---\n');

  // 2️⃣ Build prompt
  const prompt = `
You are a helpful assistant that answers questions based only on the provided PDF excerpt.
If the answer cannot be derived from the excerpt, politely say you don't know.

PDF EXCERPT:
"""
${context}
"""

Question: ${question}
Answer:
`;

  // 3️⃣ Call LLM
  const completion = await openai.createChatCompletion({
    model: 'gpt-4', // or gpt-3.5-turbo for cheaper usage
    messages: [{ role: 'user', content: prompt }],
    temperature: 0.2,
  });

  const answer = completion.data.choices[0].message?.content?.trim() ?? '';
  return answer;
}
```

## 3. Front‑end Sketch (React)

```tsx
// src/App.tsx
import { useState } from 'react';
import axios from 'axios';

export default function App() {
  const [file, setFile] = useState<File | null>(null);
  const [question, setQuestion] = useState('');
  const [answer, setAnswer] = useState('');
  const [pdfId, setPdfId] = useState<string>(''); // could be filename or generated UUID

  const uploadPdf = async () => {
    if (!file) return;
    const form = new FormData();
    form.append('file', file);
    const res = await axios.post('/upload', form);
    // assume backend returns same id we used for indexing
    setPdfId(file.name);
    alert('PDF indexed!');
  };

  const ask = async () => {
    const res = await axios.post('/ask', { question, pdfId });
    setAnswer(res.data.answer);
  };

  return (
    <div style={{ padding: 20 }}>
      <h2>PDF Chat Support</h2>

      <input type="file" accept="application/pdf" onChange={e => setFile(e.target.files?.[0] ?? null)} />
      <button onClick={uploadPdf}>Upload & Index</button>

      <hr />

      <textarea
        rows={4}
        placeholder="Ask a question about the PDF..."
        value={question}
        onChange={e => setQuestion(e.target.value)}
      />
      <button onClick={ask} disabled={!pdfId}>Ask</button>

      {answer && (
        <div style={{ marginTop: 20, border: '1px solid #ddd', padding: 10 }}>
          <strong>Answer:</strong>
          <p>{answer}</p>
        </div>
      )}
    </div>
  );
}
```

> **Tip:** In production you’ll want a proper authentication flow, per‑user PDF isolation, and cleanup of uploaded files after indexing.

## 4. Optional Enhancements

| Feature | How to Add |
|---------|------------|
| **Multiple PDFs per user** | Store a `userId` in metadata, prefix vector IDs with `userId|pdfId`. |
| **Citation / Source highlighting** | Return the chunk index + text alongside the answer; highlight in UI. |
| **Streaming responses** | Use OpenAI’s `stream: true` option and forward data via Server‑Sent Events (SSE). |
| **Local vector store (no cloud)** | Use `pgvector` with a PostgreSQL container; swap Pinecone client for pg client. |
| **Better chunking** | Use `langchain`’s `RecursiveCharacterTextSplitter` for overlap (e.g., 200‑token overlap). |
| **Document summarization** | After indexing, generate a short summary and store it as metadata for quick preview. |

## 5. Security & Cost Considerations

1. **Rate‑limit uploads & queries** – prevent abuse (`express-rate-limit`).  
2. **File validation** – only accept `application/pdf` and scan size (e.g., ≤ 10 MB).  
3. **OpenAI usage** – keep `temperature` low, limit tokens (`max_tokens`), and monitor usage via the OpenAI dashboard.  
4. **Vector DB cost** – Pinecone offers a free tier; for heavy traffic consider self‑hosted Qdrant or Faiss.
