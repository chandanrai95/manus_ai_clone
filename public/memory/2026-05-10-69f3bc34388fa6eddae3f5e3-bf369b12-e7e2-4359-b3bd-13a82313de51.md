### 2.2 Create a unified “Post” type

```ts
// src/types.ts
export interface TrendPost {
  id: string;               // unique across sources
  source: 'twitter' | 'reddit' | 'hackernews' | 'medium' | 'youtube';
  title: string;
  url: string;
  text?: string;            // raw article / tweet body
  imageUrl?: string;        // main thumbnail / preview
  videoUrl?: string;        // for YouTube shorts, etc.
  publishedAt: string;      // ISO date
}
```

### 2.3 Fetch & normalize trending content

```ts
// src/fetchers/twitter.ts
import { TrendPost } from '../types';
import { TwitterApi } from 'twitter-api-v2';

const client = new TwitterApi(process.env.TWITTER_BEARER_TOKEN!);
const roClient = client.readOnly;

export async function fetchTwitterTrending(): Promise<TrendPost[]> {
  const trends = await roClient.v2.search('#javascript', {
    max_results: 20,
    tweet_fields: ['created_at', 'public_metrics'],
    expansions: ['author_id', 'attachments.media_keys'],
    'media.fields': ['url', 'preview_image_url'],
  });

  return trends.data?.map(t => ({
    id: t.id,
    source: 'twitter',
    title: t.text.slice(0, 140),
    url: `https://twitter.com/i/web/status/${t.id}`,
    text: t.text,
    imageUrl: t.attachments?.media_keys?.[0] ? `https://pbs.twimg.com/media/${t.attachments.media_keys[0]}` : undefined,
    publishedAt: t.created_at!,
  })) ?? [];
}
```

> **Tip:** Replicate similar fetchers for Reddit (`snoowrap`), HackerNews (official API), Medium RSS (`rss-parser`), and YouTube (`googleapis`). Keep each function returning `Promise<TrendPost[]>`.

### 2.4 Encode each modality

```ts
// src/encoders.ts
import { TrendPost } from './types';
import { OpenAI } from 'openai';
import { pipeline } from '@xenova/transformers';

// Text → embedding (OpenAI embeddings)
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export async function embedText(text: string): Promise<number[]> {
  const resp = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: text,
  });
  return resp.data[0].embedding;
}

// Image → CLIP embedding (local TFJS)
let clip: any;
export async function embedImage(url: string): Promise<number[]> {
  if (!clip) {
    clip = await pipeline('feature-extraction', 'Xenova/clip-vit-base-patch32');
  }
  const img = await fetch(url).then(r => r.arrayBuffer());
  const tensor = await clip(img);
  // flatten to 1‑D Float32Array
  return Array.from(tensor.dataSync());
}

// Audio (YouTube) → Whisper → text → embed
export async function embedAudioFromYoutube(videoUrl: string): Promise<number[]> {
  // 1️⃣ download audio (ffmpeg) – omitted for brevity
  // 2️⃣ send to Whisper (OpenAI)
  const transcription = await openai.audio.transcriptions.create({
    file: /* Buffer of .mp3 */,
    model: 'whisper-1',
  });
  return embedText(transcription.text);
}
```

> **Why separate collections?**  
> • Text embeddings are dense (~1536 dims for OpenAI).  
> • CLIP image vectors are 512‑dim.  
> Storing them in distinct namespaces lets you query “text‑or‑image similarity” by merging results later.

### 2.5 Upsert into a vector store

```ts
// src/vectorStore.ts
import { PineconeClient } from '@pinecone-database/pinecone';
import { TrendPost } from './types';

const pinecone = new PineconeClient();
await pinecone.init({
  environment: process.env.PINECONE_ENV!,
  apiKey: process.env.PINECONE_API_KEY!,
});

const index = pinecone.Index('trends');

// Helper to upsert a batch
export async function upsertPosts(posts: TrendPost[]) {
  const vectors = await Promise.all(
    posts.map(async post => {
      let embedding: number[];
      if (post.text) {
        embedding = await embedText(post.text);
      } else if (post.imageUrl) {
        embedding = await embedImage(post.imageUrl);
      } else if (post.videoUrl) {
        embedding = await embedAudioFromYoutube(post.videoUrl);
      } else {
        // fallback – empty vector
        embedding = new Array(1536).fill(0);
      }

      return {
        id: post.id,
        values: embedding,
        metadata: {
          ...post,
          // keep only lightweight fields in metadata
        },
      };
    })
  );

  await index.upsert({ upsertRequest: { vectors } });
}
```

> **Cron‑style refresh** – run the fetch‑encode‑upsert pipeline every hour (e.g., via `node-cron` or a serverless function) to keep the knowledge base fresh.

### 2.6 Retrieval‑Augmented Generation (RAG)

```ts
// src/rag.ts
import { OpenAI } from 'openai';
import { PineconeClient } from '@pinecone-database/pinecone';
import { TrendPost } from './types';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
const pinecone = new PineconeClient();
await pinecone.init({ environment: process.env.PINECONE_ENV!, apiKey: process.env.PINECONE_API_KEY! });
const index = pinecone.Index('trends');

export async function answerResearchQuestion(question: string): Promise<string> {
  // 1️⃣ embed the question (same model as text)
  const qEmb = await embedText(question);

  // 2️⃣ similarity search (top‑k)
  const resp = await index.query({
    queryRequest: {
      vector: qEmb,
      topK: 5,
      includeMetadata: true,
    },
  });

  const retrieved: TrendPost[] = resp.matches?.map(m => m.metadata as TrendPost) ?? [];

  // 3️⃣ Build a prompt
  const context = retrieved
    .map((p, i) => `Source ${i + 1} (${p.source}): ${p.title}\n${p.text ?? ''}`)
    .join('\n---\n');

  const prompt = `
You are a research assistant helping Chandan write a blog post about the latest trends in JavaScript/TypeScript.

Given the user question and the following sources, produce a concise, fact‑checked answer and optionally suggest a short outline for the blog.

Question: "${question}"
Sources:
${context}
`;

  // 4️⃣ Call LLM
  const completion = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [{ role: 'user', content: prompt }],
    temperature: 0.2,
  });

  return completion.choices[0].message.content ?? '';
}
```

### 2.7 Front‑end integration (React + Radix UI)

```tsx
// ui/src/App.tsx
import { useState } from 'react';
import * as DropdownMenu from '@radix-ui/react-dropdown-menu';
import { answerResearchQuestion } from '../src/rag';

export default function App() {
  const [question, setQuestion] = useState('');
  const [answer, setAnswer] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  const ask = async () => {
    setLoading(true);
    const res = await answerResearchQuestion(question);
    setAnswer(res);
    setLoading(false);
  };

  return (
    <div className="p-8 max-w-2xl mx-auto">
      <h1 className="text-2xl font-bold mb-4">🧠 Multi‑Modal Research Assistant</h1>

      <textarea
        className="w-full p-2 border rounded mb-2"
        rows={3}
        placeholder="What do you want to know about the current JavaScript trend?"
        value={question}
        onChange={e => setQuestion(e.target.value)}
      />

      <button
        className="px-4 py-2 bg-indigo-600 text-white rounded disabled:opacity-50"
        onClick={ask}
        disabled={loading || !question.trim()}
      >
        {loading ? 'Thinking…' : 'Ask'}
      </button>

      {answer && (
        <div className="mt-6 p-4 border rounded bg-gray-50 whitespace-pre-wrap">
          {answer}
        </div>
      )}

      {/* Example: source filter using Radix Dropdown */}
      <DropdownMenu.Root>
        <DropdownMenu.Trigger asChild>
          <button className="mt-4 underline">Filter sources…</button>
        </DropdownMenu.Trigger>

        <DropdownMenu.Content className="bg-white border rounded shadow-lg p-2">
          <DropdownMenu.CheckboxItem checked={true}>Twitter</DropdownMenu.CheckboxItem>
          <DropdownMenu.CheckboxItem checked={true}>Reddit</DropdownMenu.CheckboxItem>
          <DropdownMenu.CheckboxItem checked={true}>HackerNews</DropdownMenu.CheckboxItem>
          <DropdownMenu.CheckboxItem checked={true}>Medium</DropdownMenu.CheckboxItem>
          <DropdownMenu.CheckboxItem checked={true}>YouTube</DropdownMenu.CheckboxItem>
        </DropdownMenu.Content>
      </DropdownMenu.Root>
    </div>
  );
}
```

## 5️⃣ Common Pitfalls & How to Avoid Them

| Problem | Fix |
|---------|-----|
| **Rate‑limit errors** from Twitter/Reddit | Cache responses for 5‑10 min, respect the platform’s `X-RateLimit-Remaining` header, and stagger requests. |
| **Embedding dimension mismatch** (text vs. image) | Keep separate namespaces in the vector DB, or pad/trim vectors to a common size before upserting (but separate is cleaner). |
| **Huge PDF uploads** choking your node process | Stream the file, use `pdfjs-dist` for text extraction, and chunk before embedding (you already have this for chat‑support). |
| **LLM hallucinations** when sources are sparse | Always ask the model to **cite** each fact; if a citation is missing, flag the answer for review. |
| **Cold start latency** for CLIP/Transformer models | Load the model once at server start (`await pipeline(...)`) and reuse; consider a separate micro‑service for heavy encoding. |
